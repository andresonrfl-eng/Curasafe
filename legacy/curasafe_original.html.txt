    import { getFirestore, collection, doc, setDoc, deleteDoc, onSnapshot, query } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

    // Ícones em SVG
    const SHAPES = {

      drops: `<path d="M12 3s-6 6-6 10a6 6 0 1012 0c0-4-6-10-6-10z" fill="currentColor" opacity="0.3"/><path d="M12 3s0 6-6 10a6 6 0 0012 0c0-4-6-10-6-10H12z" fill="currentColor"/>`,
      syrup: `<path d="M8 3h8v3h-2v4h3v11H7V10h3V6H8V3z" fill="currentColor" opacity="0.3"/><path d="M7 14h10v7H7v-7z" fill="currentColor"/>`,
      injection: `<path d="M14 2H10v2h1v3L9 9v11h6V9l-2-2V4h1V2z" fill="currentColor" opacity="0.3"/><path d="M11 12h2v5h-2v-5zm-2-3h6v11H9V9z" fill="currentColor"/>`
    };

    // A CORREÇÃO CRÍTICA: Gerar ID Local Imediatamente
    function getOrCreateLocalUserId() {
      let uid = localStorage.getItem('curasafe_uid');
      if (!uid) {
        // Formato CURA_XXXXXX
        uid = 'CURA_' + Math.random().toString(36).substr(2, 6).toUpperCase();
        localStorage.setItem('curasafe_uid', uid);
      }
      return uid;
    }

    const localUid = getOrCreateLocalUserId();
    const localFamily = localStorage.getItem('curasafe_familyId') || localUid;

    // Gerenciador de Estado
    window.appState = {
      uid: localUid,          // Seu ID imutável
      familyId: localFamily,  // O ID da caixa que está visualizando
      isOwnFamily: (localFamily === localUid),
      meds: JSON.parse(localStorage.getItem('curasafe_meds') || '[]'),
      logs: JSON.parse(localStorage.getItem('curasafe_logs') || '[]'),
      firebaseMedsRef: null,
      firebaseLogsRef: null,
      isFirebaseReady: false
    };

    // Atualização da UI Imediata (Tira o Carregando)
    window.updateMedfriendUI = function() {
      const codeEl = document.getElementById('myFamilyCode');
      if (codeEl) codeEl.innerText = appState.uid; // Seu código é sempre seu UID
      
      const lbl = document.getElementById('currentFamilyLabel');
      const btn = document.getElementById('btnResetFamily');
      
      if(appState.isOwnFamily) {
        lbl.innerText = "Sua Conta Pessoal";
        lbl.className = "text-xs font-bold text-brand-600 bg-brand-50 px-3 py-1 rounded-full";
        if(btn) btn.classList.add('hidden');
      } else {
        lbl.innerText = "Conectado a: " + appState.familyId;
        lbl.className = "text-xs font-bold text-purple-600 bg-purple-50 px-3 py-1 rounded-full";
        if(btn) btn.classList.remove('hidden');

    const appId = typeof __app_id !== 'undefined' ? __app_id : 'curasafe-app-v3';
    let db = null;

    async function initFirebase() {
      try {
        const fbConfig = typeof __firebase_config !== 'undefined' ? JSON.parse(__firebase_config) : null;
        if (!fbConfig) throw new Error("Sem config do Firebase no Preview");
        
        const app = initializeApp(fbConfig);
        const auth = getAuth(app);
        db = getFirestore(app);

        const token = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;
        if (token) await signInWithCustomToken(auth, token);
        else await signInAnonymously(auth);

        onAuthStateChanged(auth, (user) => {
          if (user) {
            appState.isFirebaseReady = true;
            document.getElementById('connectionStatus').classList.replace('bg-amber-400', 'bg-emerald-500');
            document.getElementById('connectionText').innerText = "Nuvem ON";
            listenToFirebase();
          }
        });
      } catch (e) {
        console.log("Rodando modo Local/Offline devido a ambiente preview restrito.");
        // O app continuará funcionando puramente via LocalStorage
      }
    }

    function listenToFirebase() {
      if (!appState.isFirebaseReady || !db) return;
      
      // Paths usando RULE 1
      const medsRef = collection(db, 'artifacts', appId, 'public', 'data', `fam_${appState.familyId}_meds`);
      const logsRef = collection(db, 'artifacts', appId, 'public', 'data', `fam_${appState.familyId}_logs`);

      onSnapshot(query(medsRef), (snapshot) => {
        appState.meds = [];
        snapshot.forEach(doc => appState.meds.push({ id: doc.id, ...doc.data() }));
        localStorage.setItem('curasafe_meds', JSON.stringify(appState.meds));
        renderAll();
      }, (err) => console.error(err));

      onSnapshot(query(logsRef), (snapshot) => {
        appState.logs = [];
        snapshot.forEach(doc => appState.logs.push({ id: doc.id, ...doc.data() }));
        localStorage.setItem('curasafe_logs', JSON.stringify(appState.logs));
        renderAll();
      }, (err) => console.error(err));
    }

    // Salvar Local & Nuvem
    window.saveMedication = async function() {
      const name = document.getElementById('inputName').value.trim();
      const dosage = document.getElementById('inputDosage').value.trim();
      const initialTime = document.getElementById('inputInitialTime').value;
      
      if (!name || !dosage) return showToast("Preencha nome e dose.", "error");

      const freq = parseInt(document.getElementById('inputFrequency').value, 10);
      const times = [];
      const [hh, mm] = initialTime.split(':').map(Number);
      const interval = 24 / freq;
      for (let i = 0; i < freq; i++) times.push(`${String(Math.floor((hh + i * interval) % 24)).padStart(2, '0')}:${String(mm).padStart(2, '0')}`);
      times.sort();

      const medData = {
        id: document.getElementById('formMedId').value || `med_${Date.now()}`,
        name, dosage, times, freq, initialTime,
        shape: document.getElementById('inputShape').value,
        color: document.getElementById('inputColor').value,
        stock: parseInt(document.getElementById('inputStock').value, 10) || 0,
        minStock: parseInt(document.getElementById('inputMinStock').value, 10) || 0,
        updatedAt: Date.now()
      };

      // Salva Local
      const idx = appState.meds.findIndex(m => m.id === medData.id);
      if(idx >= 0) appState.meds[idx] = medData; else appState.meds.push(medData);
      localStorage.setItem('curasafe_meds', JSON.stringify(appState.meds));
      
      // Tenta Nuvem
      if (appState.isFirebaseReady) {
        try {
           await setDoc(doc(db, 'artifacts', appId, 'public', 'data', `fam_${appState.familyId}_meds`, medData.id), medData);
        } catch(e){}
      }

      showToast("Salvo com sucesso!", "success");
      closeModal();
      renderAll();
    }

    window.handleDose = async function(medId, time, action) {
      if (action === 'taken') playPillSound();
      
      const dateKey = new Date().toISOString().split('T')[0];
      const logId = `${medId}_${dateKey}_${time.replace(':','')}`;
      
      const logData = { id: logId, medId, timeScheduled: time, date: dateKey, action, timestamp: Date.now() };

      // Atualiza Logs
      const lIdx = appState.logs.findIndex(l => l.id === logId);
      if(lIdx >= 0) appState.logs[lIdx] = logData; else appState.logs.push(logData);
      
      // Atualiza Estoque se Tomado
      if (action === 'taken') {
        const med = appState.meds.find(m => m.id === medId);
        if (med && med.stock > 0) {
          med.stock--;
          if (appState.isFirebaseReady) setDoc(doc(db, 'artifacts', appId, 'public', 'data', `fam_${appState.familyId}_meds`, med.id), med);
        }
      }

      localStorage.setItem('curasafe_logs', JSON.stringify(appState.logs));
      localStorage.setItem('curasafe_meds', JSON.stringify(appState.meds));

      if (appState.isFirebaseReady) {
         try { await setDoc(doc(db, 'artifacts', appId, 'public', 'data', `fam_${appState.familyId}_logs`, logId), logData); } catch(e){}
      }

      renderAll();
    }

    function renderAll() {
      renderToday();
      renderMeds();
      renderHistory();
      
      const low = appState.meds.filter(m => m.stock <= m.minStock);
      const alerts = document.getElementById('stockAlertsContainer');
      if (low.length > 0) {
        alerts.innerHTML = low.map(m => `<div class="bg-amber-100/80 border border-amber-200 text-amber-800 rounded-2xl p-3 text-xs font-bold">⚠️ Estoque de ${m.name} está baixo (${m.stock}).</div>`).join('');
        alerts.classList.replace('hidden', 'flex');
      } else {
        alerts.classList.replace('flex', 'hidden');
      }
    }

    function renderToday() {
      const quads = { morning: '', afternoon: '', evening: '', bedtime: '' };
      const dateKey = new Date().toISOString().split('T')[0];
      let tDoses = 0, tkDoses = 0;

      const todayLogs = {};
      appState.logs.forEach(l => { if (l.date === dateKey) todayLogs[`${l.medId}_${l.timeScheduled}`] = l.action; });

      appState.meds.forEach(med => {
        (med.times || []).forEach(time => {
          tDoses++;
          const status = todayLogs[`${med.id}_${time}`] || 'pending';
          if (status === 'taken') tkDoses++;
          
          const h = parseInt(time.split(':')[0], 10);
          const q = (h >= 6 && h < 12) ? 'morning' : (h >= 12 && h < 18) ? 'afternoon' : (h >= 18 && h < 23) ? 'evening' : 'bedtime';
          
          let btns = `<button onclick="handleDose('${med.id}', '${time}', 'taken')" class="flex-1 bg-brand-500 text-white font-bold text-xs py-2 rounded-xl">Tomar</button><button onclick="handleDose('${med.id}', '${time}', 'skipped')" class="px-3 py-2 bg-slate-100 text-slate-600 rounded-xl text-xs font-bold">Pular</button>`;
          if(status === 'taken') btns = `<div class="flex-1 text-center text-xs font-bold text-brand-600 py-1">Tomado ✅</div>`;
          if(status === 'skipped') btns = `<div class="flex-1 text-center text-xs font-bold text-slate-400 py-1">Pulado</div>`;

          quads[q] += `
            <div class="rounded-2xl border bg-white p-3 shadow-sm mb-2">
              <div class="flex items-center gap-3 mb-2">
                <div class="w-8 h-8 rounded-lg flex items-center justify-center" style="color:${med.color}; background:${med.color}15"><svg class="w-5 h-5" viewBox="0 0 24 24">${SHAPES[med.shape]||SHAPES.pill}</svg></div>
                <div><h5 class="font-bold text-slate-800 text-sm leading-tight">${med.name}</h5><span class="text-[10px] text-brand-600 font-bold">${time}</span></div>
              </div>
              <div class="flex gap-2">${btns}</div>
            </div>`;
        });
      });

      ['morning','afternoon','evening','bedtime'].forEach(q => {
        document.getElementById(`quad-${q}`).innerHTML = quads[q] || `<div class="text-[11px] text-slate-400 text-center py-2 font-bold uppercase">Livre</div>`;
      });

      const pct = tDoses === 0 ? 100 : Math.round((tkDoses / tDoses) * 100);
      document.getElementById('adherencePct').innerText = `${pct}%`;
      document.getElementById('adherenceCircle').style.strokeDashoffset = 213.6 - (pct / 100) * 213.6;
      document.getElementById('adherenceText').innerText = tDoses === 0 ? 'Nenhum remédio para hoje.' : `Você concluiu ${tkDoses} de ${tDoses} doses.`;
    }

             <div>

    function renderHistory() {
      const c = document.getElementById('logsContainer');

    // Audio Simples
    function playPillSound() {
      try {
        const ctx = new (window.AudioContext || window.webkitAudioContext)();
        for(let i=0; i<3; i++) {
          const osc = ctx.createOscillator(), gain = ctx.createGain();
          osc.type = 'square'; osc.frequency.value = 1000 + Math.random()*500;
          gain.gain.setValueAtTime(0.05, ctx.currentTime + i*0.05);
          gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + i*0.05 + 0.05);
          osc.connect(gain); gain.connect(ctx.destination);
          osc.start(ctx.currentTime + i*0.05); osc.stop(ctx.currentTime + i*0.05 + 0.06);
        }
      } catch(e){}
    }
